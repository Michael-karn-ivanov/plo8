"""BotAgent: any arena bot (any rl_env `Controller`) as an engine `Agent`.

On each turn it builds a one-row batch (`DecisionBatch.single`: the same
view, legal actions and mask a `TableBatch` row would have), asks the bot,
maps the action to the engine (`to_engine`, which raises on a masked action)
and submits it to the player. At the end of a hand it may rebuy up to a
configured stack (off by default). It has no `on_hole_cards` / `on_event`, so
the engine builds no views for them.
"""

def decide(self, view: TableView, legal: LegalActions) -> ActionType:
    """The engine action the bot chooses for this view and legal actions."""
    batch = DecisionBatch.single(view, legal)
    out = self.bot.act(batch)
    actions = out[0] if isinstance(out, tuple) else out
    return to_engine(int(actions[0]), legal)

def on_turn(self, player: Player, view: TableView, legal: LegalActions) -> None:
    t = self.decide(view, legal)
    if t is ActionType.FOLD:
        player.fold()
    elif t is ActionType.CHECK:
        player.check()
    elif t is ActionType.CALL:
        player.call()
    else:
        player.raise_()
def on_hand_end – rebuys up to the target stack if set (failures logged) arena/agent.py:51–61 
def on_hand_end(self, player: Player, result: HandResult) -> None:
    target = self.rebuy_to_bb
    if target is None:
        return
    missing = round(target - player.stack, 2)
    if missing <= 0:
        return
    try:
        player.rebuy(missing)
    except Exception:  # noqa: BLE001 - e.g. above the table's max buy-in
        _log.warning("rebuy of %s BB for %s failed", missing, player.name, exc_info=True)

"""Baseline bots: RandomBot, CallingStation, Maniac (RuleBot is in rules.py).

Every bot implements the rl_env `Controller` protocol (`wants_views`,

def _masked – replaces any action outside its row's mask with CHECK_CALL arena/bots.py:31–38 
def _masked(actions: Any, mask: Any) -> np.ndarray:
    """`actions` with every action outside its row's mask replaced by CHECK_CALL."""
    a = np.asarray(actions, dtype=np.int64).reshape(-1)
    m = np.asarray(mask, dtype=bool)
    if a.shape[0] == 0:
        return a
    ok = m[np.arange(a.shape[0]), a]
    return np.where(ok, a, CHECK_CALL).astype(np.int64)

def lineup_for(agent: str, opponents: list[str], n: int) -> list[Entrant]:
    """Position 0 = agent, then the opponents in order, repeated."""
    agent_name = f"{agent}*" if agent in opponents else agent
    lineup = [Entrant(agent_name, agent)]
    for i in range(n - 1):
        b = opponents[i % len(opponents)]
        lineup.append(Entrant(b, b))
    return lineup

def format_results(results: list[MatchResult]) -> list[str]:
    """The results table: one row per report name per table size."""
    rows = [COLUMNS]
    for res in results:
        for name, st in res.stats.items():
            others = ",".join(res.lineup[p] for p in range(res.table_size) if p not in st.positions)
            rows.append(
                (
                    name,
                    others,
                    str(res.table_size),
                    str(st.hands),
                    f"{st.bb_per_100:+.2f}",
                    _fmt_ci(st.ci95),
                )
            )
    widths = [max(len(r[i]) for r in rows) for i in range(len(COLUMNS))]
    right = {2, 3, 4, 5}
    out = []
    for r in rows:
        cells = [
            r[i].rjust(widths[i]) if i in right else r[i].ljust(widths[i])
            for i in range(len(COLUMNS))
        ]
        out.append("  ".join(cells).rstrip())
    return out

match.py – duplicate matches: every deal played table_size times with the lineup rotated, accounted in chips
module docstring – rotation, chip accounting and the hidden-card rule arena/match.py:1–18 
"""Duplicate matches: every deal is played `table_size` times, rotating the lineup.

A deal is one deck seed, one button and one set of per-seat stacks. Rotation
`r = 0..n-1` of deal `d` plays the same `HandSpec` with lineup position `p` at
seat `(p + r) % n` (seat `s` is played by position `(s - r) % n`). By the
engine's deal-order contract seat `s` gets the same hole cards and the board is
the same as far as it is dealt, whatever the actions, so over the rotation
group every position holds every seat's cards (and stack and position to the
button) exactly once.

Accounting is in integer chips, keyed by (deal, rotation, seat -> position),
and converted to big bets only for the report (`stats.summarize`). Zero sum is
checked per hand and for the totals; a violation raises `AccountingError`.

Only this module reads every seat's hole cards (`TableBatch.finished_hole_cards`,
for hand histories and deal-order checks), after the hand is over and before
its table is restarted. Nothing read there reaches a controller.
"""

"""RuleBot: a simple, readable tight-aggressive baseline.

All numbers live in `RULE_PARAMS` below: tuning means editing one block.

Preflop it scores its four hole cards (`preflop_score`) and raises / calls /
folds on the score. After the flop it asks the engine's evaluator
(`best_omaha`: exactly two hole cards and three board cards) for the high
category and whether it has a qualifying low, and looks up the nut low, the
nut-low draw and the nut-flush draw in tables over ranks and suits
(`postflop_class`). No hand comparison is re-implemented: the nut-low table is
built from `engine.evaluator.rank_low`.

Design decisions (spec 1.4):
- Nut low: the hand's low equals the best low any two hole cards could make on
  this board. Card removal by own hole cards is not considered; a nut low that
  may be tied (quartered) still counts as the nut.
- Nut-low draw (flop and turn): no nut low now, and some low rank not on the
  board would give the hand the nut low.
- Nut-flush draw (flop and turn): a suit with exactly two board cards and at
  least two hole cards, one of which is the highest rank of that suit not on
  the board (own cards not removed).

RuleBot reads only `batch.views[i]` (own hole cards, board, street, `raises`)
and `batch.legal[i]`; it is stateless and deterministic.
