def action_mask – (fold only when facing a bet, check-or-call always, bet-or-raise when the engine allows) rl_env/actions.py:49–61 
def action_mask(legal: LegalActions) -> tuple[bool, bool, bool]:
    """(fold, check-or-call, bet-or-raise) allowed for this decision.

    - fold: only when facing a bet (`not legal.can_check`); folding when checking
      is free is masked although the engine accepts it;
    - check-or-call: always (the engine always offers check or call to a player
      who has a decision; a violation is an engine bug and raises);
    - bet-or-raise: exactly when the engine allows a raise.
    """
    can_check = legal.can_check
    if not (can_check or legal.call_amount > 0):
        raise AssertionError(f"engine offered neither check nor call: {legal!r}")
    return (not can_check, True, bool(legal.can_raise))

def check_action – validates an action as an int 0..2 allowed by the mask rl_env/actions.py:69–82 
def check_action(action: Any, mask: Any) -> int:
    """`action` as an int in 0..2 allowed by `mask`; IllegalMaskedActionError
    otherwise (bools, non-integers, out of range, masked out)."""
    if isinstance(action, (bool, np.bool_)):
        raise IllegalMaskedActionError(f"not an action: {action!r}")
    try:
        a = operator.index(action)
    except TypeError:
        raise IllegalMaskedActionError(f"not an action: {action!r}") from None
    if not 0 <= a < N_ACTIONS:
        raise IllegalMaskedActionError(f"action must be 0, 1 or 2, got {a}")
    if not mask[a]:
        raise IllegalMaskedActionError(f"{ACTION_NAMES[a]} is masked out for this decision")
    return a

def to_engine(action: Any, legal: LegalActions) -> ActionType:
    """Masked action -> engine `ActionType` (check-or-call becomes CHECK when
    checking is free, CALL otherwise). Raises IllegalMaskedActionError for an
    action outside the mask."""
    a = check_action(action, action_mask(legal))
    if a == FOLD:
        return ActionType.FOLD
    if a == CHECK_CALL:
        return ActionType.CHECK if legal.can_check else ActionType.CALL
    return ActionType.RAISE

"""Many tables at once: decisions grouped by controller, actions submitted in
batches, finished hands reported with each seat's result.

Each seat of a hand is assigned a controller key by the caller when the hand
starts. `pending()` groups the decisions of all tables (at most one per table)
by the acting seat's controller key; a `Controller` answers a whole
`DecisionBatch` at once. Grouping and iteration orders are by table index
(never over sets or `id()`), so a run is deterministic.

`TableBatch(observations=False)` builds no observation vectors (and feeds no
tracker): for batches whose controllers all declare `wants_obs = False`
(evaluation matches with rule-based bots, the bench's `--no-obs`). Everything
else - masks, legal actions, grouping, order, views, actions and results - is
the same as with observations.

Nothing a controller receives (`DecisionBatch`) or `FinishedHand` carries any
hidden information, nor any reference to an engine object that could reach it.
The full deal of a finished hand is read separately with

def start_hand(self, table: int, spec: HandSpec, controllers: Sequence[Hashable]) -> int:
    """Start a hand on an idle table with one controller key per seat. Returns
    the hand's batch-unique, increasing id. A hand that needs no decision is
    finished at once (reported by the next `collect_finished()`). Ends the
    readability of the table's previous `finished_hole_cards`."""
    if self._running[table]:
        raise RuntimeError(f"table {table} still has a hand running")
    keys = tuple(controllers)
    if len(keys) != spec.n_seats:
        raise ValueError(f"need {spec.n_seats} controller keys, got {len(keys)}")
    tbl = self.tables[table]
    pending = tbl.start_hand(spec.n_seats, spec.stacks_bb, spec.button, spec.deck_seed)
    hand_id = self._next_hand_id
    self._next_hand_id += 1
    self._hand_id[table] = hand_id
    self._spec[table] = spec
    self._controllers[table] = keys
    self._running[table] = True
    self._acted[table] = 0
    if not pending:
        self._finish(table)
    return hand_id

def act(self, batch: Any) -> Union[np.ndarray, tuple[np.ndarray, np.ndarray]]:
    mask = np.asarray(batch.mask, dtype=bool)
    counts = mask.sum(axis=1)
    if mask.shape[0] and counts.min() == 0:
        raise ValueError("a row has no allowed action")
    u = self.rng.random(mask.shape[0])
    k = np.minimum((u * counts).astype(np.int64), counts - 1)
    cums = np.cumsum(mask, axis=1)
    actions = np.argmax(cums > k[:, None], axis=1).astype(np.int64)
    if self.with_logp:
        return actions, (-np.log(counts)).astype(np.float32)
    return actions

def critic_cards(table: Any, seat: int, out: Any = None) -> np.ndarray:
    """The other participants' hole cards relative to `seat` (see the module
    docstring). Read through `Table.privileged_hole_cards()` (the engine's
    privileged step-mode accessor)."""

def expand_critic(cards: Any) -> np.ndarray:
    """(B, 9, 4) card indices -> (B, PRIV_DIM) float32 multi-hot (a single
    (9, 4) array is treated as B = 1)."""
    arr = np.asarray(cards)
    if arr.ndim == 2:
        arr = arr[None]
    if arr.ndim != 3 or arr.shape[1:] != (N_SLOTS, 4):
        raise ValueError(f"expected (B, {N_SLOTS}, 4) card indices, got shape {arr.shape}")
    idx = arr.astype(np.int64)
    if idx.size and (idx.min() < 0 or idx.max() > NO_CARD):
        raise ValueError("card indices must be in 0..52")
    b = arr.shape[0]
    wide = np.zeros((b, N_SLOTS, NO_CARD + 1), dtype=np.float32)
    rows = np.arange(b)[:, None, None]
    slots = np.arange(N_SLOTS)[None, :, None]

    return np.ascontiguousarray(wide[:, :, :NO_CARD]).reshape(b, PRIV_DIM)

def _player_acted(self, ev: PlayerActed) -> None:
    i = self.index[ev.seat]
    amount = ev.amount_bb
    street = self.street
    self.stack[i] = ev.stack_bb
    self.street_bet[i] = ev.to_bb
    self.total[i] += amount
    self.pot += amount
    if ev.all_in:
        self.all_in[i] = True
    action = ev.action
    if action is _FOLD:
        self.folded[i] = True
    elif action is _RAISE:
        self.bets[i][street] += 1
        self.street_bets += 1
        self.aggressor = i
        if ev.to_bb > self.current_bet:
            self.current_bet = ev.to_bb
    self.acted[i] = True
    self.recent.append((i, _KIND[action], street))
