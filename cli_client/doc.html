def drain – handles every waiting inbox message cli_client/controller.py:197–209 
def drain(self) -> int:
    """Handle every message waiting in the inbox. Returns how many."""
    n = 0
    while True:
        try:
            msg = self.inbox.get_nowait()
        except queue.Empty:
            break
        self.handle_message(msg)
        n += 1
    if n:
        self.dirty = True
    return n

def tick(self, now: Optional[float] = None) -> None:
    """Timers: the hand-end default answer, the exit delay, countdown redraws."""
    now = self.clock() if now is None else now
    previous, self.now = self.now, now
    if (
        self.pending_hand_end is not None
        and self.hand_end_deadline is not None
        and now >= self.hand_end_deadline
    ):
        self.input = None
        self.confirm = None
        default = dict(self.hand_end_default)
        self._send_hand_end(**default)
        self.status = "No answer in time: " + (
            "leaving" if default.get("leave") else "sitting out"
        )
        self.dirty = True
    if self.exit_at is not None and now >= self.exit_at:
        self.done = True
    if math.floor(previous) != math.floor(now):
        self.dirty = True

def handle_message(self, msg: Any) -> None:
    self.now = self.clock()
    self.dirty = True
    if isinstance(msg, tuple):
        if msg and msg[0] == "disconnected":
            self._on_closed()
        return
    if not isinstance(msg, dict):
        return
    kind = msg.get("type")
    handler = _HANDLERS.get(kind) if isinstance(kind, str) else None
    if handler is None:
        return  # unknown message type: forward compatible
    try:
        handler(self, msg)
    except (ProtocolError, KeyError) as e:
        _log.warning("cannot handle %r message: %s", kind, e)

class ServerConnection:
    """Client -> server requests over one persistent TCP connection.

    `request(msg)` returns the response (also `ok: false` ones; a malformed
    response line comes back as `ok: false` with "bad response: ..."). A socket
    error, timeout or over-long line drops the connection, reconnects and
    resends once; if that fails too, `ConnectionLost` is raised. The server
    tells clients apart by `name`, so a new connection is fine.

def request(self, msg: dict) -> dict:
    with self._lock:
        if self._closed:
            raise ConnectionLost("the connection is closed")
        last: Optional[BaseException] = None
        for _attempt in range(2):
            try:
                if self._sock is None:
                    self._open()
                write_message(self._wfile, msg)
                response = read_message(self._rfile)
                if response is None:
                    raise ConnectionError("the server closed the connection")
                _log.debug("request %s -> %s", msg, response)
                return response
            except LineTooLongError as e:
                last = e
                self._drop()
            except ProtocolError as e:
                # One bad response line; the stream is still in sync.
                _log.warning("bad response to %s: %s", msg, e)
                return {"type": "response", "ok": False, "error": f"bad response: {e}"}
            except (OSError, ValueError) as e:
                last = e
                self._drop()
        raise ConnectionLost(str(last))

class CallbackListener:
    """Listens on `host:port` (0 = a free port; see `self.port`) for the
    server's callback connection. Binding happens in the constructor, so bind
    errors are raised there; `start()` launches the accept thread.

    A new connection replaces the old one (which ends without `DISCONNECTED`).
    Messages addressed to another player (`to`) are dropped and counted in
    `dropped`. A malformed line is logged, counted in `malformed` and skipped;
    the reader keeps going. Unknown message types are passed through. Nothing
    is put on the inbox after `close()`. All threads are daemon threads.
    """

def _accept_loop – accepts connections; a new one replaces the old cli_client/net.py:172–194 
def _accept_loop(self) -> None:
    while True:
        with self._lock:
            if self._closed:
                return
        try:
            conn, _ = self._sock.accept()
        except socket.timeout:
            continue
        except OSError:
            return
        conn.settimeout(None)
        with self._lock:
            if self._closed:
                _close(conn)
                return
            old, self._conn = self._conn, conn
            self._generation += 1
            generation = self._generation
        _close(old)
        threading.Thread(
            target=self._read_loop, args=(conn, generation), name="callback-reader", daemon=True
        ).start()
def _read_loop – reads messages, drops ones for other players, skips malformed, signals DISCONNECTED cli_client/net.py:196–231 
def _read_loop(self, conn: socket.socket, generation: int) -> None:
    f = conn.makefile("rb")
    try:
        while True:
            try:
                msg = read_message(f)
            except LineTooLongError as e:
                _log.warning("callback line too long: %s", e)
                break
            except ProtocolError as e:
                self.malformed += 1
                _log.warning("malformed callback message skipped: %s", e)
                continue
            except (OSError, ValueError):
                break
            if msg is None:
                break
            to = msg.get("to")
            if to is not None and to != self.name:
                self.dropped += 1
                continue
            with self._lock:
                closed = self._closed
            if closed:
                break  # nothing goes on the inbox after close()
            _log.debug("received %s", msg)
            self.inbox.put(msg)
    finally:
        try:
            f.close()
        except (OSError, ValueError):
            pass
        with self._lock:
            current = generation == self._generation and not self._closed
        if current:
            self.inbox.put(DISCONNECTED)

        "default": -1,
        "black": curses.COLOR_BLACK,

        "green": curses.COLOR_GREEN,

def _main(stdscr: "curses.window", client: Client, poll: float) -> int:
    try:
        curses.curs_set(0)
    except curses.error:
        pass
    curses.raw()  # Ctrl-C arrives as a key (3), not as SIGINT mid-update
    stdscr.keypad(True)
    stdscr.timeout(int(poll * 1000))
    palette = Palette()
    keymap = _keymap()
    while not client.done:
        try:
            code = stdscr.getch()
        except KeyboardInterrupt:
            code = 3
        if code == curses.KEY_RESIZE:
            client.dirty = True
        client.drain()
        if code not in (-1, curses.KEY_RESIZE):
            key = keymap.get(code) or (chr(code) if 32 <= code < 127 else None)
            if key is not None:
                client.handle_key(key)
        client.tick(time.monotonic())
        if client.bell:
            client.bell = False
            try:
                curses.beep()
            except curses.error:
                pass
        if client.dirty:
            client.dirty = False
            height, width = stdscr.getmaxyx()
            draw(stdscr, render(client.state, client.ui_view(), width, height), palette)
    return client.exit_code
