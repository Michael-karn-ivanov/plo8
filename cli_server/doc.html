"""End-to-end test mode of cli_server (`cli_server --e2e_test [--case NAME]`).

Recorded hands (`e2e_test_sets/*.json`) are replayed through a real `GameServer`
and real client processes, with a stacked deck; what every client receives is
checked against the recording.

Library `//cli_server:e2e` (public, no server, no process; other projects replay
the recorded hands through their own tables with it):
- `case`: load and validate a test-set file.
- `cases`: where the case files are; list / find / load them.
- `deck`: the engine seating of the case players and `StackedDeck`.

Runner `//cli_server:e2e_runner` (private to cli_server):
- `expect`: what every client is expected to receive.
- `client`: the scripted client process (`python -m cli_server.e2e.client`).
- `driver`: runs one case (server + clients) and all cases.
- `check`: compares client transcripts with the expectations.

This package file imports nothing, so importing a library module does not load
the runner.

def returned_chips – the uncalled bet returned: excess of the biggest contribution over the second, if its owner did not fold e2e/case.py:477–488 
def returned_chips(total: Mapping[str, int], folded: set[str]) -> dict[str, int]:
    """The uncalled bet given back: the excess of the biggest contribution over
    the second biggest, if its owner did not fold (the engine's rule)."""
    if not total:
        return {}
    by_size = sorted(total.items(), key=lambda kv: kv[1], reverse=True)
    top_id, top = by_size[0]
    second = by_size[1][1] if len(by_size) > 1 else 0
    excess = top - second
    if excess <= 0 or top_id in folded:

def encode_line(msg: dict) -> bytes:
    """One message as a newline-terminated UTF-8 JSON line."""
    if not isinstance(msg, dict) or not isinstance(msg.get("type"), str):
        raise ProtocolError(f"a message must be an object with a 'type', got {msg!r}")
    try:
        # ensure_ascii (the default) guarantees there is no raw newline inside.
        text = json.dumps(msg, separators=(",", ":"), allow_nan=False)
    except (TypeError, ValueError) as e:
        raise ProtocolError(f"cannot encode message: {e}") from None
    return (text + "\n").encode("utf-8")

def decode_line(line: Union[bytes, str]) -> dict:
    """Parse one JSON line; it must be an object with a non-empty string `type`."""
    if isinstance(line, bytes):
        try:
            line = line.decode("utf-8")
        except UnicodeDecodeError:
            raise ProtocolError("message is not valid UTF-8") from None
    try:
        obj = json.loads(line)
    except json.JSONDecodeError as e:
        raise ProtocolError(f"invalid JSON: {e.msg}") from None
    if not isinstance(obj, dict):
        raise ProtocolError("a message must be a JSON object")
    kind = obj.get("type")
    if not isinstance(kind, str) or not kind:
        raise ProtocolError("message has no 'type'")
    return obj

"""ClientSession: one joined client, used as the `conn` of the engine's `RemoteAgent`.

`RemoteAgent` only needs `send(msg)`, `recv()` and `close()`, so it keeps doing the
sequence numbers, the action timeout, dropping late replies and handling
disconnects. The session does the rest:

- `send` (called on the dealer thread) turns an engine message tuple into a JSON
  line and puts it on an outbound queue. A writer thread writes it to the client's
  callback port, so a slow client never blocks the dealer.
- `notify` queues a server-generated message (`acting`, `lobby`) the same way.
- Replies do not arrive on the callback channel. The server's request handlers
  check them against the pending `turn` / `hand_end` and put them into `inbox`,
  which `recv` reads.

  player leave.

  view)` after a `turn` is queued (turn announcements), and `on_status(session,

    """Queue a server-generated message (`acting`, `lobby`), adding `"to"`.
    Never blocks and never raises: a closed or broken session drops it.
    Returns whether the message was queued."""
    try:
        data = encode_line({**msg, "to": self.name})
    except Exception:  # noqa: BLE001
        _log.exception("cannot encode notification %r", msg)
        return False
    with self._lock:
        if self.closed:
            return False
        self._out.put(data)
    return True

def submit_action(self, seq: int, action: Union[Action, Leave]) -> None:
    """Answer to the pending `turn` with that `seq`. A rejected action does
    not use up the turn."""
    with self._lock:
        pending = self.pending_turn
        if pending is None:
            raise SessionError(f"{self.name} has no pending turn")
        if pending[0] != seq:
            raise SessionError(f"turn {seq} is not the pending turn of {self.name} ({pending[0]})")
        legal = pending[1]
        if isinstance(action, Action) and not legal.allows(action.type):
            allowed = ", ".join(t.value for t in legal.actions)
            raise SessionError(f"{action.type.value} is not legal now (legal: {allowed}, leave)")
        if not isinstance(action, (Action, Leave)):
            raise SessionError(f"not an action: {action!r}")
        self.pending_turn = None
        self.inbox.put(("action", seq, action))

def _write_loop(self) -> None:
    while True:
        item = self._out.get()
        if item is _STOP:
            break
        try:
            self._sock.sendall(item)
        except OSError as e:
            if not self._finished.is_set():
                _log.info("callback channel of %s failed: %s", self.name, e)
            self._finish(broken=True)
            return
    self._finish(broken=False)

def _monitor_loop(self) -> None:
    """The client never writes on the callback channel; reading only detects
    that it closed it."""
    while not self._finished.is_set():
        try:
            data = self._sock.recv(4096)
        except OSError:
            data = b""
        if not data:
            break
    if not self._finished.is_set():
        _log.info("callback channel of %s closed by the client", self.name)
    self._finish(broken=True)
def _finish – once: mark finished / broken, release, close the socket, wake writer and inbox session.py:312–330 
def _finish(self, broken: bool) -> None:
    with self._lock:
        if self._finished.is_set():
            return
        self._finished.set()
        if broken and not self.closed:
            self.broken = True
        self.closed = True
    self._release()
    try:
        self._sock.shutdown(socket.SHUT_RDWR)
    except OSError:
        pass
    try:
        self._sock.close()
    except OSError:
        pass
    self._out.put(_STOP)  # wake the writer if it is idle
    self.inbox.put(_EOF)

def _release(self) -> None:
    """Tell the owner (once) that this session is over."""
    with self._lock:
        if self._released:
            return
        self._released = True
    if self._on_close is not None:
        try:
            self._on_close(self)
        except Exception:  # noqa: BLE001
            _log.exception("session close callback failed")
